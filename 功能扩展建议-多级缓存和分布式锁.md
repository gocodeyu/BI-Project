**3. Service 实现（重点：分布式锁 + 多级缓存）**

```java
package com.yupi.springbootinit.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.QueryWrapper;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.github.benmanes.caffeine.cache.Cache;
import com.github.benmanes.caffeine.cache.Caffeine;
import com.yupi.springbootinit.model.entity.Chart;
import com.yupi.springbootinit.model.entity.ChartFavour;
import com.yupi.springbootinit.model.vo.ChartVO;
import com.yupi.springbootinit.service.ChartFavourService;
import com.yupi.springbootinit.service.ChartService;
import lombok.extern.slf4j.Slf4j;
import org.redisson.api.RLock;
import org.redisson.api.RedissonClient;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import javax.annotation.PostConstruct;
import javax.annotation.Resource;
import java.util.List;
import java.util.concurrent.TimeUnit;
import java.util.stream.Collectors;

@Service
@Slf4j
public class ChartFavourServiceImpl implements ChartFavourService {
    
    @Resource
    private ChartFavourMapper chartFavourMapper;
    
    @Resource
    private ChartService chartService;
    
    @Resource
    private RedissonClient redissonClient;
    
    @Resource
    private StringRedisTemplate stringRedisTemplate;
    
    // 多级缓存：Caffeine 本地缓存
    private Cache<String, Page<ChartVO>> caffeineCache;
    
    private static final String REDIS_KEY_PREFIX = "bi:chart:favour:list:";
    private static final String LOCK_KEY_PREFIX = "lock:chart:favour:";
    private static final int REDIS_TTL_MINUTES = 30;
    private static final int CAFFEINE_TTL_SECONDS = 60;
    
    @PostConstruct
    public void init() {
        caffeineCache = Caffeine.newBuilder()
                .maximumSize(500)
                .expireAfterWrite(CAFFEINE_TTL_SECONDS, TimeUnit.SECONDS)
                .build();
    }
    
    /**
     * 收藏/取消收藏（使用分布式锁防止并发问题）
     */
    @Override
    @Transactional(rollbackFor = Exception.class)
    public int doChartFavour(long chartId, long userId) {
        // 分布式锁Key：防止同一用户并发操作
        String lockKey = LOCK_KEY_PREFIX + chartId + ":" + userId;
        RLock lock = redissonClient.getLock(lockKey);
        
        try {
            // 尝试加锁，最多等待3秒，锁定后10秒自动释放
            if (lock.tryLock(3, 10, TimeUnit.SECONDS)) {
                try {
                    // 查询是否已收藏
                    QueryWrapper<ChartFavour> queryWrapper = new QueryWrapper<>();
                    queryWrapper.eq("chartId", chartId);
                    queryWrapper.eq("userId", userId);
                    queryWrapper.eq("isDelete", 0);
                    ChartFavour oldChartFavour = chartFavourMapper.selectOne(queryWrapper);
                    
                    if (oldChartFavour != null) {
                        // 已收藏，执行取消收藏
                        boolean result = chartFavourMapper.deleteById(oldChartFavour.getId());
                        if (result) {
                            // 删除用户收藏列表缓存
                            evictUserFavourCache(userId);
                            log.info("取消收藏成功: chartId={}, userId={}", chartId, userId);
                            return -1; // 返回-1表示取消收藏
                        }
                    } else {
                        // 未收藏，执行收藏
                        ChartFavour chartFavour = new ChartFavour();
                        chartFavour.setChartId(chartId);
                        chartFavour.setUserId(userId);
                        int result = chartFavourMapper.insert(chartFavour);
                        if (result > 0) {
                            // 删除用户收藏列表缓存
                            evictUserFavourCache(userId);
                            log.info("收藏成功: chartId={}, userId={}", chartId, userId);
                            return 1; // 返回1表示收藏成功
                        }
                    }
                } finally {
                    // 释放锁
                    if (lock.isHeldByCurrentThread()) {
                        lock.unlock();
                    }
                }
            } else {
                log.warn("获取分布式锁失败: lockKey={}", lockKey);
                throw new BusinessException(ErrorCode.SYSTEM_ERROR, "操作过于频繁，请稍后重试");
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            log.error("获取分布式锁被中断", e);
            throw new BusinessException(ErrorCode.SYSTEM_ERROR, "操作失败");
        } catch (Exception e) {
            log.error("收藏操作失败: chartId={}, userId={}", chartId, userId, e);
            throw new BusinessException(ErrorCode.SYSTEM_ERROR, "操作失败");
        }
        
        return 0;
    }
    
    /**
     * 获取用户收藏列表（多级缓存：Caffeine → Redis → DB）
     */
    @Override
    public Page<ChartVO> listMyFavourChartByPage(long current, long size, long userId) {
        String cacheKey = REDIS_KEY_PREFIX + userId + ":p" + current + ":s" + size;
        String caffeineKey = "favour:" + userId + ":p" + current + ":s" + size;
        
        try {
            // 1. 查 Caffeine 本地缓存
            Page<ChartVO> result = caffeineCache.getIfPresent(caffeineKey);
            if (result != null) {
                log.debug("Caffeine 缓存命中: userId={}, current={}", userId, current);
                return result;
            }
            
            // 2. 查 Redis
            String redisValue = stringRedisTemplate.opsForValue().get(cacheKey);
            if (StringUtils.isNotBlank(redisValue)) {
                try {
                    Gson gson = new Gson();
                    Page<ChartVO> cachedPage = gson.fromJson(redisValue, new TypeToken<Page<ChartVO>>(){}.getType());
                    if (cachedPage != null) {
                        // 写入 Caffeine
                        caffeineCache.put(caffeineKey, cachedPage);
                        log.debug("Redis 缓存命中: userId={}, current={}", userId, current);
                        return cachedPage;
                    }
                } catch (Exception e) {
                    log.error("Redis 缓存反序列化失败", e);
                }
            }
            
            // 3. 查 DB
            Page<ChartFavour> favourPage = new Page<>(current, size);
            QueryWrapper<ChartFavour> queryWrapper = new QueryWrapper<>();
            queryWrapper.eq("userId", userId);
            queryWrapper.eq("isDelete", 0);
            queryWrapper.orderByDesc("createTime");
            Page<ChartFavour> chartFavourPage = chartFavourMapper.selectPage(favourPage, queryWrapper);
            
            // 转换为 ChartVO
            Page<ChartVO> chartVOPage = new Page<>(chartFavourPage.getCurrent(), 
                    chartFavourPage.getSize(), chartFavourPage.getTotal());
            List<ChartVO> chartVOList = chartFavourPage.getRecords().stream()
                    .map(favour -> {
                        Chart chart = chartService.getById(favour.getChartId());
                        return ChartVO.objToVo(chart);
                    })
                    .filter(vo -> vo != null)
                    .collect(Collectors.toList());
            chartVOPage.setRecords(chartVOList);
            
            // 4. 写入缓存
            if (chartVOPage != null && !chartVOPage.getRecords().isEmpty()) {
                Gson gson = new Gson();
                String json = gson.toJson(chartVOPage);
                stringRedisTemplate.opsForValue().set(cacheKey, json, 
                        REDIS_TTL_MINUTES, TimeUnit.MINUTES);
                caffeineCache.put(caffeineKey, chartVOPage);
            }
            
            return chartVOPage;
            
        } catch (Exception e) {
            log.error("获取收藏列表失败: userId={}", userId, e);
            // 降级：直接查 DB
            return queryFromDb(current, size, userId);
        }
    }
    
    /**
     * 判断用户是否已收藏
     */
    @Override
    public boolean isFavour(long chartId, long userId) {
        QueryWrapper<ChartFavour> queryWrapper = new QueryWrapper<>();
        queryWrapper.eq("chartId", chartId);
        queryWrapper.eq("userId", userId);
        queryWrapper.eq("isDelete", 0);
        return chartFavourMapper.selectCount(queryWrapper) > 0;
    }
    
    /**
     * 删除用户收藏列表缓存
     */
    private void evictUserFavourCache(Long userId) {
        if (userId == null || userId <= 0) {
            return;
        }
        try {
            // 删除 Redis 中该用户的所有收藏列表缓存（使用模式匹配）
            Set<String> keys = stringRedisTemplate.keys(REDIS_KEY_PREFIX + userId + ":*");
            if (keys != null && !keys.isEmpty()) {
                stringRedisTemplate.delete(keys);
            }
            // 删除 Caffeine 缓存（简化处理，删除所有相关key）
            caffeineCache.invalidateAll();
            log.debug("删除用户收藏列表缓存: userId={}", userId);
        } catch (Exception e) {
            log.error("删除用户收藏列表缓存失败: userId={}", userId, e);
        }
    }
    
    private Page<ChartVO> queryFromDb(long current, long size, long userId) {
        // 直接查 DB 的逻辑
        // ...（同上）
    }
}
```

#### 核心代码实现

```java
package com.yupi.springbootinit.service.impl;

import com.github.benmanes.caffeine.cache.Cache;
import com.github.benmanes.caffeine.cache.Caffeine;
import com.yupi.springbootinit.model.entity.Chart;
import com.yupi.springbootinit.model.entity.ChartThumb;
import com.yupi.springbootinit.service.ChartThumbService;
import lombok.extern.slf4j.Slf4j;
import org.redisson.api.RLock;
import org.redisson.api.RedissonClient;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import javax.annotation.PostConstruct;
import javax.annotation.Resource;
import java.util.concurrent.TimeUnit;

@Service
@Slf4j
public class ChartThumbServiceImpl implements ChartThumbService {
    
    @Resource
    private ChartThumbMapper chartThumbMapper;
    
    @Resource
    private ChartService chartService;
    
    @Resource
    private RedissonClient redissonClient;
    
    @Resource
    private StringRedisTemplate stringRedisTemplate;
    
    // 点赞数缓存（Caffeine → Redis → DB）
    private Cache<Long, Integer> thumbNumCache;
    
    private static final String REDIS_THUMB_NUM_KEY = "bi:chart:thumb:num:";
    private static final String LOCK_KEY_PREFIX = "lock:chart:thumb:";
    private static final int REDIS_TTL_HOURS = 24;
    private static final int CAFFEINE_TTL_MINUTES = 10;
    
    @PostConstruct
    public void init() {
        thumbNumCache = Caffeine.newBuilder()
                .maximumSize(1000)
                .expireAfterWrite(CAFFEINE_TTL_MINUTES, TimeUnit.MINUTES)
                .build();
    }
    
    /**
     * 点赞/取消点赞（使用分布式锁）
     */
    @Override
    @Transactional(rollbackFor = Exception.class)
    public int doChartThumb(long chartId, long userId) {
        // 分布式锁：防止并发点赞导致数据不一致
        String lockKey = LOCK_KEY_PREFIX + chartId + ":" + userId;
        RLock lock = redissonClient.getLock(lockKey);
        
        try {
            if (lock.tryLock(3, 10, TimeUnit.SECONDS)) {
                try {
                    // 查询是否已点赞
                    QueryWrapper<ChartThumb> queryWrapper = new QueryWrapper<>();
                    queryWrapper.eq("chartId", chartId);
                    queryWrapper.eq("userId", userId);
                    queryWrapper.eq("isDelete", 0);
                    ChartThumb oldChartThumb = chartThumbMapper.selectOne(queryWrapper);
                    
                    if (oldChartThumb != null) {
                        // 已点赞，执行取消点赞
                        chartThumbMapper.deleteById(oldChartThumb.getId());
                        // 更新点赞数（减1）
                        updateThumbNum(chartId, -1);
                        evictThumbNumCache(chartId);
                        return -1;
                    } else {
                        // 未点赞，执行点赞
                        ChartThumb chartThumb = new ChartThumb();
                        chartThumb.setChartId(chartId);
                        chartThumb.setUserId(userId);
                        chartThumbMapper.insert(chartThumb);
                        // 更新点赞数（加1）
                        updateThumbNum(chartId, 1);
                        evictThumbNumCache(chartId);
                        return 1;
                    }
                } finally {
                    if (lock.isHeldByCurrentThread()) {
                        lock.unlock();
                    }
                }
            } else {
                throw new BusinessException(ErrorCode.SYSTEM_ERROR, "操作过于频繁");
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new BusinessException(ErrorCode.SYSTEM_ERROR, "操作失败");
        }
    }
    
    /**
     * 获取图表点赞数（多级缓存）
     */
    @Override
    public int getThumbNum(long chartId) {
        try {
            // 1. 查 Caffeine
            Integer thumbNum = thumbNumCache.getIfPresent(chartId);
            if (thumbNum != null) {
                return thumbNum;
            }
            
            // 2. 查 Redis
            String redisKey = REDIS_THUMB_NUM_KEY + chartId;
            String redisValue = stringRedisTemplate.opsForValue().get(redisKey);
            if (StringUtils.isNotBlank(redisValue)) {
                thumbNum = Integer.parseInt(redisValue);
                thumbNumCache.put(chartId, thumbNum);
                return thumbNum;
            }
            
            // 3. 查 DB
            Chart chart = chartService.getById(chartId);
            thumbNum = chart != null ? chart.getThumbNum() : 0;
            
            // 4. 写入缓存
            if (thumbNum != null) {
                stringRedisTemplate.opsForValue().set(redisKey, String.valueOf(thumbNum),
                        REDIS_TTL_HOURS, TimeUnit.HOURS);
                thumbNumCache.put(chartId, thumbNum);
            }
            
            return thumbNum != null ? thumbNum : 0;
            
        } catch (Exception e) {
            log.error("获取点赞数失败: chartId={}", chartId, e);
            // 降级：直接查 DB
            Chart chart = chartService.getById(chartId);
            return chart != null && chart.getThumbNum() != null ? chart.getThumbNum() : 0;
        }
    }
    
    /**
     * 更新点赞数（使用分布式锁防止并发）
     */
    private void updateThumbNum(long chartId, int delta) {
        String lockKey = LOCK_KEY_PREFIX + "num:" + chartId;
        RLock lock = redissonClient.getLock(lockKey);
        
        try {
            if (lock.tryLock(3, 5, TimeUnit.SECONDS)) {
                try {
                    Chart chart = chartService.getById(chartId);
                    if (chart != null) {
                        int newThumbNum = (chart.getThumbNum() != null ? chart.getThumbNum() : 0) + delta;
                        chart.setThumbNum(Math.max(0, newThumbNum)); // 确保不为负数
                        chartService.updateById(chart);
                    }
                } finally {
                    if (lock.isHeldByCurrentThread()) {
                        lock.unlock();
                    }
                }
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        } catch (Exception e) {
            log.error("更新点赞数失败: chartId={}, delta={}", chartId, delta, e);
        }
    }
    
    private void evictThumbNumCache(long chartId) {
        try {
            String redisKey = REDIS_THUMB_NUM_KEY + chartId;
            stringRedisTemplate.delete(redisKey);
            thumbNumCache.invalidate(chartId);
        } catch (Exception e) {
            log.error("删除点赞数缓存失败: chartId={}", chartId, e);
        }
    }
}
```

```java
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;

import javax.annotation.PostConstruct;
import javax.annotation.Resource;
import java.util.List;
import java.util.concurrent.TimeUnit;
import java.util.stream.Collectors;

@Service
@Slf4j
public class ChartRankServiceImpl implements ChartRankService {

    // 排行榜缓存

    private static final String REDIS_RANK_KEY = "bi:chart:rank:hot:";
    private static final String LOCK_KEY = "lock:chart:rank:rebuild";
    private static final int REDIS_TTL_MINUTES = 30;
    private static final int CAFFEINE_TTL_MINUTES = 5;
    
    @PostConstruct
    public void init() {
        rankCache = Caffeine.newBuilder()
                .maximumSize(10) // 只缓存几种排行榜
                .expireAfterWrite(CAFFEINE_TTL_MINUTES, TimeUnit.MINUTES)
                .build();
        
        // 系统启动时预热排行榜
        warmUpRankCache();
    }
    
    /**
     * 获取热门图表排行榜（多级缓存 + 分布式锁防击穿）
     */
    @Override
    public List<ChartVO> getHotChartRank(int limit) {
        String cacheKey = "hot:" + limit;
        String redisKey = REDIS_RANK_KEY + limit;
        
        try {
            // 1. 查 Caffeine
            List<ChartVO> result = rankCache.getIfPresent(cacheKey);
            if (result != null) {
                log.debug("Caffeine 缓存命中: limit={}", limit);
                return result;
            }
            
            // 2. 查 Redis
            String redisValue = stringRedisTemplate.opsForValue().get(redisKey);
            if (StringUtils.isNotBlank(redisValue)) {
                try {
                    Gson gson = new Gson();
                    result = gson.fromJson(redisValue, new TypeToken<List<ChartVO>>(){}.getType());
                    if (result != null) {
                        rankCache.put(cacheKey, result);
                        log.debug("Redis 缓存命中: limit={}", limit);
                        return result;
                    }
                } catch (Exception e) {
                    log.error("Redis 缓存反序列化失败", e);
                }
            }
            
            // 3. 缓存未命中，使用分布式锁防止缓存击穿
            RLock lock = redissonClient.getLock(LOCK_KEY);
            try {
                // 双重检查：获取锁后再次检查缓存
                redisValue = stringRedisTemplate.opsForValue().get(redisKey);
                if (StringUtils.isNotBlank(redisValue)) {
                    Gson gson = new Gson();
                    result = gson.fromJson(redisValue, new TypeToken<List<ChartVO>>(){}.getType());
                    if (result != null) {
                        rankCache.put(cacheKey, result);
                        return result;
                    }
                }
                
                // 尝试加锁
                if (lock.tryLock(5, 30, TimeUnit.SECONDS)) {
                    try {
                        // 从 DB 查询
                        result = queryHotChartsFromDb(limit);
                        
                        // 写入缓存
                        if (result != null && !result.isEmpty()) {
                            Gson gson = new Gson();
                            String json = gson.toJson(result);
                            stringRedisTemplate.opsForValue().set(redisKey, json,
                                    REDIS_TTL_MINUTES, TimeUnit.MINUTES);
                            rankCache.put(cacheKey, result);
                        }
                        
                        return result;
                    } finally {
                        if (lock.isHeldByCurrentThread()) {
                            lock.unlock();
                        }
                    }
                } else {
                    // 获取锁失败，降级：直接查 DB（避免长时间等待）
                    log.warn("获取排行榜锁失败，降级查DB: limit={}", limit);
                    return queryHotChartsFromDb(limit);
                }
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                return queryHotChartsFromDb(limit);
            }
            
        } catch (Exception e) {
            log.error("获取热门排行榜失败: limit={}", limit, e);
            return queryHotChartsFromDb(limit);
        }
    }
    
    /**
     * 从 DB 查询热门图表（按点赞数、收藏数、创建时间综合排序）
     */
    private List<ChartVO> queryHotChartsFromDb(int limit) {
        // 使用 SQL 查询：综合点赞数、收藏数、创建时间计算热度
        // SELECT * FROM chart 
        // WHERE isDelete = 0 AND status = 'succeed'
        // ORDER BY (thumbNum * 2 + favourNum + TIMESTAMPDIFF(HOUR, createTime, NOW())) DESC
        // LIMIT ?
        
        QueryWrapper<Chart> queryWrapper = new QueryWrapper<>();
        queryWrapper.eq("isDelete", 0);
        queryWrapper.eq("status", "succeed");
        queryWrapper.orderByDesc("updateTime"); // 简化：按更新时间排序
        queryWrapper.last("LIMIT " + limit);
        
        List<Chart> charts = chartService.list(queryWrapper);
        return charts.stream()
                .map(ChartVO::objToVo)
                .collect(Collectors.toList());
    }
    
    /**
     * 定时任务：每30分钟更新一次排行榜（避免频繁查询DB）
     */
    @Scheduled(fixedRate = 30 * 60 * 1000) // 30分钟
    public void refreshHotRank() {
        log.info("开始刷新热门排行榜");
        try {
            // 删除缓存，触发重建
            for (int limit : new int[]{10, 20, 50}) {
                String redisKey = REDIS_RANK_KEY + limit;
                stringRedisTemplate.delete(redisKey);
                rankCache.invalidate("hot:" + limit);
            }
            // 预热缓存
            warmUpRankCache();
            log.info("热门排行榜刷新完成");
        } catch (Exception e) {
            log.error("刷新热门排行榜失败", e);
        }
    }
    
    /**
     * 预热排行榜缓存
     */
    private void warmUpRankCache() {
        try {
            for (int limit : new int[]{10, 20}) {
                getHotChartRank(limit);
            }
            log.info("排行榜缓存预热完成");
        } catch (Exception e) {
            log.error("排行榜缓存预热失败", e);
        }
    }
}
```

1. **读多写少的数据**：收藏列表、点赞数、排行榜
2. **热点数据**：热门图表、统计数据
3. **减少DB压力**：通过缓存减少数据库查询

### 缓存更新策略
1. **写库后删除缓存**：保证数据一致性
2. **定时刷新**：排行榜等定期更新的数据
