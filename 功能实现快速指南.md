**ChartFavourController.java**
```java
package com.yupi.springbootinit.controller;

import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.yupi.springbootinit.common.BaseResponse;
import com.yupi.springbootinit.common.ResultUtils;
import com.yupi.springbootinit.model.vo.ChartListVO;
import com.yupi.springbootinit.service.ChartFavourService;
import lombok.extern.slf4j.Slf4j;
import org.springframework.web.bind.annotation.*;

import javax.annotation.Resource;
import javax.servlet.http.HttpServletRequest;

@RestController
@RequestMapping("/chart/favour")
@Slf4j
public class ChartFavourController {

    @Resource
    private ChartFavourService chartFavourService;

    @Resource
    private UserService userService;

    /**
     * 收藏/取消收藏
     */
    @PostMapping("/do")
    public BaseResponse<Integer> doChartFavour(@RequestParam long chartId, 
                                                HttpServletRequest request) {
        User loginUser = userService.getLoginUser(request);
        int result = chartFavourService.doChartFavour(chartId, loginUser.getId());
        return ResultUtils.success(result);
    }

    /**
     * 获取我的收藏列表
     */
    @GetMapping("/my/list")
    public BaseResponse<Page<ChartListVO>> listMyFavourChartByPage(
            @RequestParam(defaultValue = "1") long current,
            @RequestParam(defaultValue = "10") long size,
            HttpServletRequest request) {
        User loginUser = userService.getLoginUser(request);
        Page<ChartListVO> result = chartFavourService.listMyFavourChartByPage(
                current, size, loginUser.getId());
        return ResultUtils.success(result);
    }

    /**
     * 判断是否已收藏
     */
    @GetMapping("/is")
    public BaseResponse<Boolean> isFavour(@RequestParam long chartId,
                                          HttpServletRequest request) {
        User loginUser = userService.getLoginUser(request);
        boolean result = chartFavourService.isFavour(chartId, loginUser.getId());
        return ResultUtils.success(result);
    }
}
```

## 🔑 关键技术点

### 1. 分布式锁使用

```java
// 获取锁
String lockKey = "lock:chart:favour:" + chartId + ":" + userId;
RLock lock = redissonClient.getLock(lockKey);

try {
    if (lock.tryLock(3, 10, TimeUnit.SECONDS)) {
        try {
            // 业务逻辑
        } finally {
            if (lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }
    }
} catch (InterruptedException e) {
    Thread.currentThread().interrupt();
}
```

### 2. 多级缓存使用

```java
// 1. 查 Caffeine
result = caffeineCache.getIfPresent(key);
if (result != null) return result;

// 2. 查 Redis
String redisValue = stringRedisTemplate.opsForValue().get(redisKey);
if (StringUtils.isNotBlank(redisValue)) {
    result = parseFromJson(redisValue);
    caffeineCache.put(key, result);
    return result;
}

// 3. 查 DB
result = queryFromDb();
// 写入缓存
stringRedisTemplate.opsForValue().set(redisKey, toJson(result), ttl, TimeUnit.MINUTES);
caffeineCache.put(key, result);
return result;
```

### 3. 防止缓存击穿

```java
// 使用分布式锁防止缓存击穿
RLock lock = redissonClient.getLock("lock:chart:rank:rebuild");
if (lock.tryLock(5, 30, TimeUnit.SECONDS)) {
    try {
        // 双重检查
        if (缓存存在) return 缓存;
        
        // 查 DB 并写入缓存
        result = queryFromDb();
        writeToCache(result);
    } finally {
        lock.unlock();
    }
}
```

## 📝 注意事项

1. **分布式锁超时时间**：根据业务耗时设置，避免死锁
2. **缓存TTL**：根据数据更新频率设置，避免数据过期
3. **异常处理**：缓存失败时降级到DB查询
4. **日志记录**：记录关键操作，便于排查问题
5. **事务保证**：写库操作要加事务注解
